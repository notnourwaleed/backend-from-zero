Project skeleton

index.js
const express = require('express');
const cookieParser = require('cookie-parser');
const userRouter = require('./routes/userRoutes');

const app = express();
app.use(express.json());
app.use(cookieParser());

app.use('/users', userRouter);

module.exports = { app };
server.js
const { app } = require('./index.js');
const { db, createUserTable } = require('./db.js');
const dotenv = require('dotenv');
dotenv.config();

async function startServer() {
  try {
    await db.query(createUserTable);
    app.listen(process.env.PORT, process.env.HOSTNAME, () => {
      console.log(`Server is running at http://${process.env.HOSTNAME}:${process.env.PORT}`);
    });
  } catch (error) {
    console.error('Error initializing the database:', error);
    process.exitCode = 1;
    await db.end();
  }
}

startServer();

Controller patterns

Get one (validate id → query → 404 → 200)
const getOne = async (req, res) => {
  const id = Number(req.params.id);
  if (!Number.isSafeInteger(id) || id <= 0) {
    return res.status(400).json({ message: 'Invalid ID' });
  }
  try {
    const [rows] = await db.query('SELECT * FROM THINGS WHERE ID = ?', [id]);
    if (rows.length === 0) return res.status(404).json({ message: 'Not found' });
    return res.status(200).json({ data: rows[0] });
  } catch (error) {
    console.error(error);
    return res.status(500).json({ message: 'Database error' });
  }
};
Create (validate body → insert → select new row → 201)
const createOne = async (req, res) => {
  const { name, price } = req.body;
  if (!name || price === undefined) {
    return res.status(400).json({ message: 'Please provide all required fields' });
  }
  try {
    const [result] = await db.query('INSERT INTO THINGS (NAME, PRICE) VALUES (?, ?)', [name, price]);
    const [rows] = await db.query('SELECT * FROM THINGS WHERE ID = ?', [result.insertId]);
    return res.status(201).json({ message: 'Created', data: rows[0] });
  } catch (error) {
    if (error.code === 'ER_DUP_ENTRY') return res.status(409).json({ message: 'Already exists' });
    console.error(error);
    return res.status(500).json({ message: 'Database error' });
  }
};
Partial update
const [rows] = await db.query('SELECT * FROM THINGS WHERE ID = ?', [id]);
if (rows.length === 0) return res.status(404).json({ message: 'Not found' });
const newName = name || rows[0].NAME;
await db.query('UPDATE THINGS SET NAME = ? WHERE ID = ?', [newName, id]);
Delete
const [result] = await db.query('DELETE FROM THINGS WHERE ID = ?', [id]);
if (result.affectedRows === 0) return res.status(404).json({ message: 'Not found' });
return res.status(200).json({ message: 'Deleted' });

Status codes

200 OK · 201 Created · 400 bad/missing input · 401 wrong login · 403 not allowed · 404 not found · 409 already exists · 500 server/database error
